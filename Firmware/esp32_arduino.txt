#include <RadioLib.h>
#include <SPI.h>

#define LORA_MISO 3
#define LORA_MOSI 6
#define LORA_SCK  5
#define LORA_CS   7
#define LORA_RST  8
#define LORA_DIO1 33
#define LORA_BUSY 34

const int RELAY_PIN = 48;

#define RELAY_OPEN  HIGH
#define RELAY_CLOSE LOW

const uint32_t IDLE_INTERVAL_S = 60;   // heartbeat with closed valve (set to 600 after testing)
const uint32_t WATERING_INTERVAL_S  = 120;
const uint32_t DEFAULT_WATERING_MIN = 10;
const uint32_t MAX_WATERING_MIN     = 30;

uint64_t joinEUI = 0x0000000000000000;
uint64_t devEUI  = 0x0000000000000000;
uint8_t appKey[16] = { 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00,
                       0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00 };
uint8_t nwkKey[16] = { 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00,
                       0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00, 0x00 };

SX1262 radio = new Module(LORA_CS, LORA_DIO1, LORA_RST, LORA_BUSY);
LoRaWANNode node(&radio, &EU868);

bool isOpen = false;
unsigned long closeAt = 0;

void closeValve() {
  digitalWrite(RELAY_PIN, RELAY_CLOSE);
  isOpen = false;
  closeAt = 0;
}

void openValve(uint32_t minutes) {
  if (minutes == 0) minutes = DEFAULT_WATERING_MIN;
  if (minutes > MAX_WATERING_MIN) minutes = MAX_WATERING_MIN;
  digitalWrite(RELAY_PIN, RELAY_OPEN);
  isOpen = true;
  closeAt = millis() + minutes * 60000UL;
  Serial.printf(">>> COMMAND RECEIVED: OPEN VALVE %lu min <<<\n", (unsigned long)minutes);
}

void checkTimer() {
  if (isOpen && (long)(millis() - closeAt) >= 0) {
    closeValve();
    Serial.println("Timer finished: VALVE CLOSED");
  }
}

void setup() {
  Serial.begin(115200);
  delay(2000);

  pinMode(RELAY_PIN, OUTPUT);
  digitalWrite(RELAY_PIN, RELAY_CLOSE);

  SPI.begin(LORA_SCK, LORA_MISO, LORA_MOSI, LORA_CS);

  Serial.print("Starting SX1262... ");
  int16_t state = radio.begin();
  if (state != RADIOLIB_ERR_NONE) {
    Serial.printf("failed, code %d\n", state);
    while (true) delay(1000);
  }
  Serial.println("OK");
  radio.setDio2AsRfSwitch(true);

  node.beginOTAA(joinEUI, devEUI, nwkKey, appKey);

  Serial.println("Connecting to TTN...");
  state = node.activateOTAA();
  while (state != RADIOLIB_LORAWAN_NEW_SESSION && state != RADIOLIB_LORAWAN_SESSION_RESTORED) {
    Serial.printf("Join failed (%d), retrying in 30 s\n", state);
    delay(30000);
    state = node.activateOTAA();
  }
  Serial.println("Connected to TTN!");
}

void loop() {
  uint8_t remaining = 0;
  if (isOpen) {
    long ms = (long)(closeAt - millis());
    remaining = (ms > 0) ? (uint8_t)((ms + 59999) / 60000) : 0;
  }
  uint8_t up[2] = { (uint8_t)(isOpen ? 1 : 0), remaining };
  uint8_t down[16];
  size_t len = 0;

  Serial.println("Sending heartbeat status...");
  int16_t state = node.sendReceive(up, sizeof(up), 1, down, &len);

  if (state > 0 && len > 0) {
    if (down[0] == 0x01) {
      openValve(len >= 2 ? down[1] : 0);
    } else if (down[0] == 0x00) {
      closeValve();
      Serial.println(">>> COMMAND RECEIVED: CLOSE VALVE <<<");
    }
  } else if (state != RADIOLIB_ERR_NONE && state <= 0) {
    Serial.printf("Send error: %d\n", state);
  }
  Serial.println("Heartbeat sent. Waiting for commands...");

  unsigned long waitTime = (isOpen ? WATERING_INTERVAL_S : IDLE_INTERVAL_S) * 1000UL;
  unsigned long t0 = millis();
  while (millis() - t0 < waitTime) {
    checkTimer();
    delay(200);
  }
}
